## Installation

```ts
import { CalendarContextProvider } from "@/components/application/date-picker/calendar";
import { CalendarCell } from "@/components/application/date-picker/cell";
import { DatePicker } from "@/components/application/date-picker/date-picker";
import { DateRangePicker } from "@/components/application/date-picker/date-range-picker";
import { RangeCalendarContextProvider } from "@/components/application/date-picker/range-calendar";
```

## Usage

```tsx
<DatePicker value={date} onChange={setDate} onApply={save} />
<DateRangePicker value={range} onChange={setRange} />
```

Picking a date updates the value immediately (so `onChange` fires as the user clicks). **Cancel** restores the value the picker had when it opened and fires `onChange` with it; **Apply** just closes.
