## Installation

```ts
import { Avatar } from "@/components/base/avatar/avatar";
import { AvatarLabelGroup } from "@/components/base/avatar/avatar-label-group";
import { AvatarProfilePhoto } from "@/components/base/avatar/avatar-profile-photo";
```

## Usage

```tsx
<Avatar src={user.avatarUrl} alt={user.name} initials="OR" size="md" />

// Logos and other non-square marks: show the whole image instead of cropping it.
<Avatar src={school.logoUrl} alt={school.name} imageFit="contain" />
```
